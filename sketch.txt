function startHandposeLoop() {
  if (!handposeModel) return;

  handposeModel.on('predict', predictions => {
    const palmOpen = detectOpenPalm(predictions);

    if (currentState === State.TITLE) {
      handleTitlePalm(palmOpen);
    } else if (currentState === State.WAITING_PALM) {
      handleDialoguePalm(palmOpen);
    } else if (currentState === State.WAITING_OBJ) {
      // palm can still be used as fallback (optional)
    }

    // diary pages use swipe — handled separately
  });
}

function detectOpenPalm(predictions) {
  if (!predictions || predictions.length === 0) return false;
  const hand = predictions[0];
  const landmarks = hand.landmarks;

  // Check that all fingertips are above their base knuckles
  // Fingertip indices: 4(thumb), 8(index), 12(middle), 16(ring), 20(pinky)
  // MCP knuckle indices: 2, 5, 9, 13, 17
  const tips  = [8, 12, 16, 20];
  const bases = [5,  9, 13, 17];

  let openCount = 0;
  for (let i = 0; i < tips.length; i++) {
    if (landmarks[tips[i]][1] < landmarks[bases[i]][1]) openCount++;
  }
  return openCount >= 3; // at least 3 fingers extended
}

function detectSwipeLeft(predictions, prevLandmarks) {
  if (!predictions || predictions.length === 0) return false;
  if (!prevLandmarks) return false;
  const hand = predictions[0];
  const wristX     = hand.landmarks[0][0];
  const prevWristX = prevLandmarks[0][0];
  return (wristX - prevWristX) > 55; // moved right in mirrored view = swipe left
}

function handleTitlePalm(palmOpen) {
  if (palmOpen) {
    if (!palmHoldStart) palmHoldStart = Date.now();
    if (Date.now() - palmHoldStart >= PALM_HOLD_MS) {
      palmHoldStart = null;
      startGame();
    }
  } else {
    palmHoldStart = null;
  }
}

async function loadChapter(chapterIndex, fromBeat = 0) {
  const chapter = GAME_SCRIPT[chapterIndex];
  currentChapter = chapterIndex;

  // Update room background with fade
  await fadeRoomTransition(async () => {
    $roomBg.style.backgroundImage = `url('${ROOM_IMAGES[chapter.room]}')`;
    $roomLabel.textContent = chapter.label;
    $roomLabel.classList.remove('visible');
  });

  await sleep(200);
  $roomLabel.classList.add('visible');

  currentBeat = fromBeat;
}

async function loadBeat(beatIndex) {
  const chapter = GAME_SCRIPT[currentChapter];
  if (beatIndex >= chapter.beats.length) {
    // Move to next chapter
    if (currentChapter + 1 < GAME_SCRIPT.length) {
      await loadChapter(currentChapter + 1);
      await loadBeat(0);
    } else {
      endGame();
    }
    return;
  }

  currentBeat = beatIndex;
  const beat = chapter.beats[beatIndex];

  // Fade out current text
  $dialogueText.classList.remove('visible');
  await sleep(400);

  // Style text for diary entries
  if (beat.type === 'diary') {
    $dialogueText.classList.add('diary');
  } else {
    $dialogueText.classList.remove('diary');
  }

  $dialogueText.textContent = beat.text;

  // Show/hide gesture prompt
  $gesturePrompt.classList.remove('hidden');
  $gesturePrompt.classList.remove('visible');

  if (beat.type === 'object') {
    // Object recognition beat
    showObjectPrompt(beat);
    $gesturePrompt.classList.add('hidden');
    currentState = State.WAITING_OBJ;
    startObjectDetection(beat.object);
  } else if (beat.type === 'diary') {
    // Diary page — swipe gesture
    $gesturePrompt.textContent = '👋 Swipe hand to turn page';
    currentState = State.WAITING_PALM;
    startDiarySwipeDetection();
    hideObjectPrompt();
  } else if (beat.type === 'ending') {
    hideObjectPrompt();
    currentState = State.WAITING_PALM;
  } else {
    // Regular dialogue beat
    hideObjectPrompt();
    currentState = State.WAITING_PALM;
  }

  // Fade in text
  await sleep(100);
  $dialogueText.classList.add('visible');

  if (beat.type !== 'object') {
    await sleep(500);
    $gesturePrompt.classList.add('visible');
  }
}

// ─────────────────────────────────────────────────────
// DIALOGUE PALM (advance to next beat)
// ─────────────────────────────────────────────────────

function handleDialoguePalm(palmOpen) {
  if (currentState !== State.WAITING_PALM) return;

  if (palmOpen) {
    if (!palmHoldStart) palmHoldStart = Date.now();
    // Visual feedback — pulse the gesture prompt
    if (Date.now() - palmHoldStart >= PALM_HOLD_MS) {
      palmHoldStart = null;
      advanceBeat();
    }
  } else {
    palmHoldStart = null;
  }
}

function advanceBeat() {
  const chapter = GAME_SCRIPT[currentChapter];
  const beat = chapter.beats[currentBeat];

  if (beat.type === 'ending') {
    endGame();
    return;
  }

  currentState = State.TRANSITIONING;
  loadBeat(currentBeat + 1).then(() => {
    if (currentState === State.TRANSITIONING) {
      currentState = State.WAITING_PALM;
    }
  });
}

function showObjectPrompt(beat) {
  $objectName.textContent = beat.instruction || beat.object;
  $objectPrompt.classList.remove('hidden');
  $detectionRing.classList.remove('detected');
  $detectionLabel.textContent = 'Waiting…';
}

function hideObjectPrompt() {
  $objectPrompt.classList.add('hidden');
  $camContainer.classList.remove('detected');
}

async function startObjectDetection(targetClass) {
  if (!tmModel) {
    // No model loaded — skip after 2 seconds (dev mode)
    console.warn('No TM model — skipping object detection for:', targetClass);
    await sleep(2000);
    onObjectDetected();
    return;
  }

  let frameId;

  async function detect() {
    if (currentState !== State.WAITING_OBJ) return;

    try {
      const predictions = await tmModel.predict($webcam);
      const best = predictions.reduce((a, b) => a.probability > b.probability ? a : b);

      const matched = best.className.toLowerCase() === targetClass.toLowerCase()
                   && best.probability >= CONFIDENCE_THRESHOLD;

      if (matched) {
        if (!detectionHoldStart) detectionHoldStart = Date.now();
        const elapsed = Date.now() - detectionHoldStart;
        const pct = Math.min(elapsed / DETECTION_HOLD_MS, 1);

        $detectionRing.classList.add('detected');
        $camContainer.classList.add('detected');
        $detectionLabel.textContent = 'Recognized ✓';

        if (elapsed >= DETECTION_HOLD_MS) {
          detectionHoldStart = null;
          cancelAnimationFrame(frameId);
          onObjectDetected();
          return;
        }
      } else {
        detectionHoldStart = null;
        $detectionRing.classList.remove('detected');
        $camContainer.classList.remove('detected');
        $detectionLabel.textContent = 'Waiting…';
      }
    } catch (err) {
      console.error('Detection error:', err);
    }

    frameId = requestAnimationFrame(detect);
  }

  detect();
}

function onObjectDetected() {
  hideObjectPrompt();
  currentState = State.TRANSITIONING;
  loadBeat(currentBeat + 1).then(() => {

function startDiarySwipeDetection() {
  if (!handposeModel) return;

  handposeModel.on('predict', predictions => {
    if (currentState !== State.WAITING_PALM) return;

    const beat = GAME_SCRIPT[currentChapter].beats[currentBeat];
    if (beat.type !== 'diary') return;
    if (swipeCooldown) return;

    const swipe = detectSwipeLeft(predictions, prevHandLandmarks);
    if (predictions && predictions.length > 0) {
      prevHandLandmarks = predictions[0].landmarks;
    }

    if (swipe) {
      swipeCooldown = true;
      setTimeout(() => swipeCooldown = false, 800);
      advanceBeat();
    }
  });
}
